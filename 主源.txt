local function httpGet(url,headers)
 if USE_PROXY then
  local proxyUrl=PROXY_CONFIG.ROOT.."/"..PROXY_CONFIG.ACCESS_KEY.."/"..HttpService:UrlEncode(url)
  local resp=HttpService:GetAsync(proxyUrl)
  local ok,data=pcall(HttpService.JSONDecode,HttpService,resp)
  if ok and data and data.body then return data.body end
  return resp
 else
  return HttpService:GetAsync(url)
 end
end

 end
end

local function probePath(origin,dir,dist,params)
 local nearest=nil
 for i,h in ipairs(HEIGHTS)do
  local o=origin+Vector3.new(0,h,0)
  local hit=WS:Raycast(o,dir*dist,params)
  if hit and(not nearest or hit.Distance<nearest)then nearest=hit.Distance end
  if i==2 then
   local right=dir:Cross(Vector3.new(0,1,0))
   if right.Magnitude>0.001 then right=right.Unit
    for _,side in ipairs({1,-1})do local o2=o+right*(NC.LAT_OFFSET*side)local hit2=WS:Raycast(o2,dir*dist,params)if hit2 and(not nearest or hit2.Distance<nearest)then nearest=hit2.Distance end end
   end
  end
 end
 return nearest
end
local function setCrouch(hum,on)
 if not hum then return end

local function findGap(origin,dir,params,hum)
 local probe=origin local travelled=0 local sawLowHole=false
 for _=1,NC.MAX_LAYER do
  local remain=NC.MAX_PROBE-travelled
  if remain<=0.1 then return nil,sawLowHole end
  local hitDist=probePath(probe,dir,remain,params)
  if not hitDist then return probe,sawLowHole end
  local headHit=WS:Raycast(probe+Vector3.new(0,1.5,0),dir*remain,params)
  local bodyHit=WS:Raycast(probe,dir*remain,params)
  if headHit and not bodyHit then sawLowHole=true end
  local gap=math.clamp(NC.GAP_MIN+hitDist*NC.GAP_SLOPE,NC.GAP_MIN,NC.GAP_MAX)
  local step=hitDist+gap
  probe=probe+dir*step
  travelled=travelled+step
 end
 return nil,sawLowHole
end

 elseif mode=="Third"then local back=tHead.CFrame.LookVector*-(5+offset)goal=CFrame.lookAt(tHead.Position+back+Vector3.new(0,1.5,0),tHead.Position)
